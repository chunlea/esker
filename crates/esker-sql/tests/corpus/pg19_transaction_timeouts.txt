# `lock_timeout`, `statement_timeout`, `idle_in_transaction_session_timeout`, deadlock detection
# and serialization failure — as a real PostgreSQL 19beta1 answers them.
#
# **Written because run 47 hung on the file that tests these.**
# `adapters/postgresql/transaction_test.rb` sat for twenty minutes: it asks the server to time a
# wait out, this node never does, and the pass stalled on one `ESTABLISHED` connection with both
# sides at 0% CPU. The file is a server unit, and this is its oracle.
#
# **THREE columns will not do, and neither will one session.** `sesscap.py` drives one psql session
# and `bindcap.rb` one libpq connection; a lock wait, a `lock_timeout` firing and a deadlock all
# need a *second* session to hold the conflicting lock. `scratchpad/txncap.rb` drives two
# connections, A and B, and two threads where they must block on each other. Format:
#
#   <session> <tab> <sql> <tab> <types> <tab> <rows | !SQLSTATE message>
#
# Lines whose session column is `#` are section markers, not statements.
#
# **Volatile values are normalised.** A deadlock DETAIL names live backend pids and transaction ids,
# which differ every run; they are rewritten to `N` so that what is pinned is the message *shape*.
# Two consecutive runs of the tool are byte-identical with that normalisation and not without it.
#
# The table is created and dropped by the capture itself, and the final `SELECT count(*)` answers
# `0`. Sessions A and B are separate connections, so this one is not wrapped in a single
# `BEGIN … ROLLBACK` — B has to see A's committed rows for a lock to conflict at all.
#
# THINGS REASONING WOULD GET WRONG
#
#   * **Five behaviours, five different SQLSTATEs, and two of them share one code.**
#       `57014` statement timeout — `canceling statement due to statement timeout`
#       `55P03` lock timeout      — `canceling statement due to lock timeout`
#       `55P03` **NOWAIT**        — `could not obtain lock on row in relation "tt_rows"`
#       `40P01` deadlock detected
#       `40001` could not serialize access due to read/write dependencies among transactions
#     `lock_timeout` and `FOR UPDATE NOWAIT` are the same class with different sentences, so a node
#     cannot distinguish them by SQLSTATE alone — the message is the specification.
#   * **`statement_timeout` also ends a lock wait**, and then the error is `57014`, not `55P03`. The
#     same `UPDATE` blocked by the same lock reports differently depending on which timeout fired,
#     which is exactly what the test file checks.
#   * **`idle_in_transaction_session_timeout` terminates the SESSION, not the statement.** After it
#     fires the connection is gone: the next statement is a libpq-level
#     `FATAL: terminating connection due to idle-in-transaction timeout`, and the `ROLLBACK` after
#     it cannot even find a socket. It is not a cancelled statement on a live session.
#   * **The deadlock DETAIL is two sentences describing both sides**, naming each process, the
#     `ShareLock`, and the transaction it waits on, plus `HINT: See server log for query details.`
#     Only **one** of the two transactions is killed — the other's `UPDATE` succeeds — so a node that
#     aborts both is wrong in a way a single-session test cannot see.
#   * **`FOR UPDATE SKIP LOCKED` returns the rows it could take rather than erroring**: with row 1
#     held by B it answers `2`, the other row, immediately.
#   * **A serialization failure surfaces at `COMMIT`, not at the conflicting statement.** Both
#     `SELECT sum(n)` and both `INSERT`s succeed; A's `COMMIT` succeeds; B's `COMMIT` is `40001` with
#     `DETAIL: Reason code: Canceled on identification as a pivot, during commit attempt.` and a
#     retry HINT. Afterwards the table has 3 rows, not 4 — B's insert is gone.
#   * **`SET LOCAL` scopes a timeout to the transaction**, which is how the tests set one without
#     leaking it into the session — and `SET statement_timeout = 0` is how it is turned off, not
#     `RESET` alone.
A	DROP TABLE IF EXISTS tt_rows		-
A	CREATE TABLE tt_rows (id bigint primary key, n integer)		-
A	INSERT INTO tt_rows VALUES (1, 10), (2, 20)		-
#	statement_timeout: one session is enough		
A	SET statement_timeout = '150ms'		-
A	SELECT pg_sleep(2)		!57014 canceling statement due to statement timeout
A	SET statement_timeout = 0		-
A	SELECT 'alive'		alive
#	lock_timeout: B holds the row, A waits and gives up		
B	BEGIN		-
B	UPDATE tt_rows SET n = 11 WHERE id = 1		-
A	BEGIN		-
A	SET LOCAL lock_timeout = '150ms'		-
A	UPDATE tt_rows SET n = 12 WHERE id = 1		!55P03 canceling statement due to lock timeout
A	ROLLBACK		-
#	statement_timeout also ends a lock wait, with a different SQLSTATE		
A	BEGIN		-
A	SET LOCAL statement_timeout = '150ms'		-
A	UPDATE tt_rows SET n = 13 WHERE id = 1		!57014 canceling statement due to statement timeout
A	ROLLBACK		-
#	NOWAIT and SKIP LOCKED do not wait at all		
A	BEGIN		-
A	SELECT id FROM tt_rows WHERE id = 1 FOR UPDATE NOWAIT		!55P03 could not obtain lock on row in relation "tt_rows"
A	ROLLBACK		-
A	BEGIN		-
A	SELECT id FROM tt_rows ORDER BY id FOR UPDATE SKIP LOCKED		2
A	ROLLBACK		-
B	ROLLBACK		-
#	deadlock: A takes row 1 then row 2, B takes row 2 then row 1; PostgreSQL kills one		
A	BEGIN		-
B	BEGIN		-
A	UPDATE tt_rows SET n = n + 1 WHERE id = 1		-
B	UPDATE tt_rows SET n = n + 1 WHERE id = 2		-
A	UPDATE tt_rows SET n = n + 1 WHERE id = 2  (concurrent)		!40P01 deadlock detected DETAIL: process N waits for ShareLock on transaction N; blocked by process N. process N waits for ShareLock on transaction N; blocked by process N. HINT: See server log for query details.
B	UPDATE tt_rows SET n = n + 1 WHERE id = 1  (concurrent)		-
A	ROLLBACK		-
B	ROLLBACK		-
#	serialization failure needs SERIALIZABLE on both sides		
A	BEGIN ISOLATION LEVEL SERIALIZABLE		-
B	BEGIN ISOLATION LEVEL SERIALIZABLE		-
A	SELECT sum(n) FROM tt_rows		30
B	SELECT sum(n) FROM tt_rows		30
A	INSERT INTO tt_rows VALUES (3, 30)		-
B	INSERT INTO tt_rows VALUES (4, 40)		-
A	COMMIT		-
B	COMMIT		!40001 could not serialize access due to read/write dependencies among transactions DETAIL: Reason code: Canceled on identification as a pivot, during commit attempt. HINT: The transaction might succeed if retried.
A	SELECT count(*) FROM tt_rows		3
#	idle_in_transaction_session_timeout ends the SESSION, not the statement		
B	SET idle_in_transaction_session_timeout = '150ms'		-
B	BEGIN		-
B	SELECT 1		1
B	SELECT 'after idling'		!????? PQconsumeInput() FATAL:  terminating connection due to idle-in-transaction timeout
B	ROLLBACK		!????? PQsocket() can't get socket descriptor
A	DROP TABLE IF EXISTS tt_rows		-
A	SELECT count(*) FROM pg_class WHERE relname = 'tt_rows'		0
