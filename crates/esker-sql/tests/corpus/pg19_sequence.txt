# What a real PostgreSQL 19beta1 does with a `bigserial` and an identity column -- unit 2 of
# `docs/plans/phase-9-rails.md`, and what `Model.create!` sits on.
#
# Nothing here was written from the documentation. Every line was put to the running `esker-pg19`
# container on 2026-09-01, **in this order**, and what is recorded is what came back. The replay
# is stateful: a `CREATE TABLE` here is a table the rest of the file uses and an `INSERT` is a row
# the next line sees, which is most of what a sequence is about.
#
#   statement <tab> types <tab> rows      the result, and the types `\gdesc` declared for it
#   statement <tab> <tab> -               a command with no result set, only a tag
#   statement <tab> !SQLSTATE message     PostgreSQL refused it; DETAIL and HINT follow
#
# THE THREE A READER WOULD HAVE GOT WRONG:
#
#   * `GENERATED BY DEFAULT AS IDENTITY` takes an explicit value **and does not advance the
#     sequence** -- so `z3` below hands out 1, then takes 50, then hands out 2, and the collision
#     that sets up is a real server's behaviour and not a bug in ours.
#   * only `GENERATED ALWAYS` refuses one, and it refuses with `428C9` -- a class with exactly this
#     condition in it -- naming the column, saying it is GENERATED ALWAYS, and hinting at the
#     clause that overrides it.
#   * the sequence's **name is a relation name**: `CREATE TABLE z1_id_seq` is `42P07`, and
#     `DROP TABLE` over it is `42809` with a hint naming `DROP SEQUENCE` rather than `DROP INDEX`.
#     One namespace, three kinds of thing in it.
#
# `serial` and `smallserial` are accepted there and are `0A000` here: they are `integer` and
# `smallint` under other names, this crate has neither, and an `int8` would take every value a real
# server answers `22003` for (ADR 0031, `docs/plans/phase-9-rails.md` §2 unit 2).
#
# --- what a bigserial and an identity column are, and what they do with a value the user wrote
CREATE TABLE z1 (id bigserial PRIMARY KEY, n text)		-
CREATE TABLE z2 (id int8 GENERATED ALWAYS AS IDENTITY PRIMARY KEY, n text)		-
CREATE TABLE z3 (id int8 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, n text)		-
INSERT INTO z1 (n) VALUES ('a') RETURNING id	bigint	1
INSERT INTO z1 (n) VALUES ('b'), ('c') RETURNING id	bigint	2 ; 3
INSERT INTO z1 (id, n) VALUES (100, 'explicit') RETURNING id	bigint	100
INSERT INTO z1 (n) VALUES ('d') RETURNING id	bigint	4
INSERT INTO z2 (n) VALUES ('a') RETURNING id	bigint	1
INSERT INTO z2 (id, n) VALUES (50, 'b')	!428C9 cannot insert a non-DEFAULT value into column "id" DETAIL: Column "id" is an identity column defined as GENERATED ALWAYS. HINT: Use OVERRIDING SYSTEM VALUE to override.
INSERT INTO z3 (n) VALUES ('a') RETURNING id	bigint	1
INSERT INTO z3 (id, n) VALUES (50, 'b') RETURNING id	bigint	50
INSERT INTO z3 (n) VALUES ('c') RETURNING id	bigint	2
SELECT id, n FROM z1 ORDER BY id	bigint,text	1|a ; 2|b ; 3|c ; 4|d ; 100|explicit
SELECT column_name, is_nullable FROM information_schema.columns WHERE table_name = 'z2' ORDER BY ordinal_position	name,character varying(3)	id|NO ; n|YES
CREATE TABLE z1_id_seq (a int8)	!42P07 relation "z1_id_seq" already exists
CREATE INDEX z1_id_seq ON z1 (n)	!42P07 relation "z1_id_seq" already exists
DROP TABLE z1_id_seq	!42809 "z1_id_seq" is not a table HINT: Use DROP SEQUENCE to remove a sequence.
DROP INDEX z1_id_seq	!42809 "z1_id_seq" is not an index HINT: Use DROP SEQUENCE to remove a sequence.
ALTER TABLE z1_id_seq ADD COLUMN q int8	!42809 ALTER action ADD COLUMN cannot be performed on relation "z1_id_seq" DETAIL: This operation is not supported for sequences.
