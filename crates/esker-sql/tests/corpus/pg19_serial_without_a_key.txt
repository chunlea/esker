# A `serial` or identity column on a table with no key — debt #93 — as a real PostgreSQL 19beta1 answers it.
#
# Captured 2026-09-13 by lane s2-plpgsql with `sesscap.py`, in two sessions, each inside `BEGIN … ROLLBACK`, every
# refusal in its own `SAVEPOINT` / `ROLLBACK TO`, in a schema of its own; each session's last line proves it left
# nothing behind: `esker-coord/s2-d93a.stmts` (what an `INSERT` that leaves the column out takes, `RETURNING`,
# `currval`, the column later in the row, `BY DEFAULT` and `ALWAYS`, beside a `UNIQUE` only, and added to a table that
# already has rows) and `s2-d93b.stmts` (what the catalog says about the column: `pg_get_serial_sequence`, the
# default, `attidentity`, and what `DROP COLUMN` takes with it). The `\gdesc` pass runs on a connection of its own,
# where those schemas do not exist, so a statement naming one of them has an empty declared-types column.
#
#!escaped
#
#   statement <tab> types <tab> rows      the result, and the types `\gdesc` declared for it
#   statement <tab> <tab> -               a command with no result set, only a tag
#   statement <tab> !SQLSTATE message     PostgreSQL refused it; DETAIL and HINT follow
#
# THINGS REASONING WOULD GET WRONG
#
#   * **A key is not what makes a serial work.** `CREATE TABLE ser (id serial, v text)` and an `INSERT` that leaves
#     `id` out is `1`, then `2`, `3` — the same as with a primary key; a `UNIQUE` alone, a column that is not first, an
#     identity of either kind: all the same.
#   * **A value the statement writes does not consume a number**: after `id = 100`, the next is `5`, not `101`.
#   * **`pg_get_serial_sequence` names the sequence for its own column and for no other**, schema-qualified, and
#     `DROP COLUMN` takes the sequence with it.
#
BEGIN		-
CREATE SCHEMA s2sq		-
SET search_path = s2sq, public		-
CREATE TABLE ser (id serial, v text)		-
INSERT INTO ser (v) VALUES ('x') RETURNING id, v		1|x
INSERT INTO ser (v) VALUES ('y'), ('z') RETURNING id, v		2|y ; 3|z
INSERT INTO ser VALUES (DEFAULT, 'w') RETURNING id, v		4|w
INSERT INTO ser (id, v) VALUES (100, 'explicit') RETURNING id, v		100|explicit
INSERT INTO ser (v) VALUES ('after') RETURNING id, v		5|after
SELECT 'r', currval('ser_id_seq')		r|5
SELECT id, v FROM ser ORDER BY id		1|x ; 2|y ; 3|z ; 4|w ; 5|after ; 100|explicit
UPDATE ser SET v = 'updated' WHERE id = 1 RETURNING id, v		1|updated
CREATE TABLE ser_last (v text, id bigserial)		-
INSERT INTO ser_last (v) VALUES ('x'), ('y') RETURNING v, id		x|1 ; y|2
CREATE TABLE ident (id integer GENERATED ALWAYS AS IDENTITY, v text)		-
INSERT INTO ident (v) VALUES ('x') RETURNING id, v		1|x
SAVEPOINT a		-
INSERT INTO ident (id, v) VALUES (5, 'no') RETURNING id, v	!428C9 cannot insert a non-DEFAULT value into column "id" DETAIL: Column "id" is an identity column defined as GENERATED ALWAYS. HINT: Use OVERRIDING SYSTEM VALUE to override.
ROLLBACK TO a		-
CREATE TABLE identd (id bigint GENERATED BY DEFAULT AS IDENTITY, v text)		-
INSERT INTO identd (v) VALUES ('x') RETURNING id, v		1|x
INSERT INTO identd (id, v) VALUES (9, 'nine') RETURNING id, v		9|nine
INSERT INTO identd (v) VALUES ('y') RETURNING id, v		2|y
CREATE TABLE uniq (id serial UNIQUE, v text)		-
INSERT INTO uniq (v) VALUES ('x'), ('y') RETURNING id, v		1|x ; 2|y
CREATE TABLE later (v text)		-
INSERT INTO later VALUES ('a'), ('b')		-
SAVEPOINT l		-
ALTER TABLE later ADD COLUMN id serial		-
ROLLBACK TO l		-
RESET search_path		-
ROLLBACK		-
SELECT count(*) FROM pg_namespace WHERE nspname = 's2sq'	bigint	0
# the second session: what the catalog says about the column
BEGIN		-
CREATE SCHEMA s2sq2		-
SET search_path = s2sq2, public		-
CREATE TABLE ser (v text, id serial, w text)		-
SELECT 'r', pg_get_serial_sequence('ser', 'id'), pg_get_serial_sequence('ser', 'v')	text,text,text	r|s2sq2.ser_id_seq|\\N
SELECT 'r', column_name, column_default, is_nullable FROM information_schema.columns WHERE table_name = 'ser' ORDER BY ordinal_position	text,name,character varying,character varying(3)	r|v|\\N|YES ; r|id|nextval('ser_id_seq'::regclass)|NO ; r|w|\\N|YES
SELECT 'r', attname, attnotnull, attidentity FROM pg_attribute WHERE attrelid = 'ser'::regclass AND attnum > 0 ORDER BY attnum		r|v|f| ; r|id|t| ; r|w|f|
CREATE TABLE ident (v text, id bigint GENERATED BY DEFAULT AS IDENTITY)		-
SELECT 'r', pg_get_serial_sequence('ident', 'id')	text,text	r|s2sq2.ident_id_seq
SELECT 'r', attname, attnotnull, attidentity FROM pg_attribute WHERE attrelid = 'ident'::regclass AND attnum > 0 ORDER BY attnum		r|v|f| ; r|id|t|d
INSERT INTO ser (v, w) VALUES ('a', 'b') RETURNING v, id, w		a|1|b
SELECT 'r', nextval('ser_id_seq')		r|2
ALTER TABLE ser DROP COLUMN id		-
SELECT 'r', count(*) FROM pg_class WHERE relname = 'ser_id_seq' AND relnamespace = 's2sq2'::regnamespace		r|0
SELECT v, w FROM ser		a|b
RESET search_path		-
ROLLBACK		-
