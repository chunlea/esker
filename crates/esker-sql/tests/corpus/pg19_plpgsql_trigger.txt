# Row triggers and their PL/pgSQL functions — the subset ADR 0113 builds, and the shapes just
# outside it — as a real PostgreSQL 19beta1 answers them.
#
# Captured 2026-09-13 by lane s2-plpgsql with `sesscap.py`, in **three sessions**, each inside
# `BEGIN … ROLLBACK` with every refusal in its own `SAVEPOINT` / `ROLLBACK TO`. The second runs in
# a schema of its own: the oracle's `public` already holds the suite's
# `pk_autopopulated_by_a_trigger_records` and `postgresql_partitioned_table_parent` from a schema
# load, and the census's statements are captured here verbatim, so they need names nobody has.
# The third, in a schema of its own too, asks what the first two left open: `NEW` and `OLD` where
# the event leaves them null, generated columns, and the foreign-key actions that are not a
# `CASCADE` of a delete. Each session's last line proves it left nothing behind. The two schema
# sessions set `search_path` with a plain `SET`, which the `ROLLBACK` undoes on both servers:
# `SET LOCAL` is refused here by name, and the abort it caused swallowed everything after it.
#
#!escaped
#
#   statement <tab> types <tab> rows      the result, and the types `\gdesc` declared for it
#   statement <tab> <tab> -               a command with no result set, only a tag
#   statement <tab> !SQLSTATE message     PostgreSQL refused it; DETAIL and HINT follow
#
# THINGS REASONING WOULD GET WRONG
#
#   * **`BEFORE` triggers on one event fire in name order, and each sees the NEW the one before it
#     returned**: `s2t_a_upper` then `s2t_b_skip`, so the skipped row was upper-cased first.
#   * **`RETURN NULL` from a `BEFORE` trigger removes the row from the statement** — not written,
#     not in `RETURNING`, not counted: statement 762's inserts answer `INSERT 0 0` and no row.
#   * **An `AFTER` trigger sees every row its statement wrote**: both of a two-row insert count five
#     rows, not four and five.
#   * **`DISABLE TRIGGER USER` and `ALL` both flip every user trigger's `tgenabled` to `D`**, and
#     `ENABLE TRIGGER USER` after `DISABLE TRIGGER ALL` turns them all back on.
#   * **A foreign key's `ON DELETE CASCADE` fires the child's row triggers**, and an `UPDATE` through
#     an inheritance parent fires the triggers of the child the row is in.
#   * **A trigger body's errors are PostgreSQL's**: `2F005` for a body that ends without `RETURN`,
#     `42804` for `RETURN 1`, `42703 record "new" has no field "nosuch"`, `54001` with its HINT for a
#     trigger that fires itself — and `42P13` for a `LANGUAGE sql` function returning `trigger`,
#     `42P17` for a trigger naming a function that does not return one.
#   * **A trigger's writes are its statement's**: the audit row a trigger wrote for a statement that
#     then failed on a duplicate key is not there.
#   * **`OLD` in an `INSERT` trigger and `NEW` in a `DELETE` one read `NULL`, and returned as they
#     are they are `RETURN NULL`**: the insert writes nothing and the delete keeps its row. Assigning
#     one of its fields makes it a row — `NEW.id = 7; RETURN NEW` in a `DELETE` trigger deletes.
#   * **A generated column reads `NULL` in a `BEFORE UPDATE` trigger too**, not the value stored:
#     it is computed after the triggers, from the row they leave.
#   * **`SET NULL` and `ON UPDATE CASCADE` fire the child's `UPDATE` triggers.** A child's
#     `BEFORE DELETE` answering `RETURN NULL` under `ON DELETE CASCADE` keeps the child row, with no
#     error; a child's `BEFORE UPDATE` that points the key somewhere else is `23503`.
#   * **`ENABLE TRIGGER USER` after `DISABLE TRIGGER ALL` leaves the foreign key's own triggers
#     off**: a row with no parent still goes in.

CREATE TRIGGER g_t BEFORE INSERT OR UPDATE ON g FOR EACH ROW EXECUTE FUNCTION g_see()		-
INSERT INTO g (id) VALUES (1) RETURNING id, doubled		2|4
UPDATE g SET id = 10 RETURNING id, doubled		11|22
SELECT 'r', note FROM a WHERE note LIKE 'g %' ORDER BY seq		r|g sees null ; r|g sees null

# ON UPDATE CASCADE and ON DELETE SET NULL fire the child's UPDATE triggers
CREATE TABLE p (id integer PRIMARY KEY)		-
CREATE TABLE c (id integer PRIMARY KEY, p integer REFERENCES p ON DELETE SET NULL ON UPDATE CASCADE)		-
CREATE FUNCTION c_touch() RETURNS trigger AS $$ BEGIN INSERT INTO a (note) VALUES ('c update ' || OLD.id || ' ' || coalesce(OLD.p::text, 'null') || ' to ' || coalesce(NEW.p::text, 'null')); RETURN NEW; END $$ LANGUAGE plpgsql		-
CREATE TRIGGER c_t BEFORE UPDATE ON c FOR EACH ROW EXECUTE FUNCTION c_touch()		-
INSERT INTO p VALUES (1), (2)		-
INSERT INTO c VALUES (10, 1), (20, 2)		-
UPDATE p SET id = 3 WHERE id = 2		-
DELETE FROM p WHERE id = 1		-
SELECT 'r', id, p FROM c ORDER BY id		r|10|\\N ; r|20|3
SELECT 'r', note FROM a WHERE note LIKE 'c %' ORDER BY seq		r|c update 20 2 to 3 ; r|c update 10 1 to null
# a child's BEFORE UPDATE trigger that sends ON UPDATE CASCADE somewhere else
CREATE FUNCTION c_elsewhere() RETURNS trigger AS $$ BEGIN NEW.p = 99; RETURN NEW; END $$ LANGUAGE plpgsql		-
CREATE TRIGGER c_u BEFORE UPDATE ON c FOR EACH ROW EXECUTE FUNCTION c_elsewhere()		-
SAVEPOINT p6		-

ROLLBACK TO p6		-

# DISABLE TRIGGER ALL then ENABLE TRIGGER USER leaves the foreign key's own triggers off
