# A **chain** of joins — the rung-3 blocker, and what a left-deep fold has to get right
#
# `ActiveRecord`'s `indexes()` sends four tables and three joins, `pg_class` twice under two
# aliases, all `ON`, mixing `INNER` and `LEFT`. Rung 3 stopped on it for four scoreboard runs with
# `0A000 more than one JOIN is not supported`.
#
#   statement <tab> types <tab> rows      the result, and the types `\gdesc` declared for it
#   statement <tab> <tab> -               a command with no result set, only a tag
#   statement <tab> !SQLSTATE message     PostgreSQL refused it
#
# THE ONE THAT DECIDES WHETHER A FOLD IS A JOIN PLANNER OR A GUESS:
#
#   `ja LEFT JOIN jb … JOIN jc …` returns **one** row, not four. The inner join is applied to the
#   already-NULL-extended rows and throws them back out. A chain planned as "each join against the
#   original left table" would keep all four, and a chain that reordered the steps would keep two.
#   Left-deep, in the order written, is not an optimisation choice here — it is the semantics.
#
# The four combinations are all here (INNER-INNER, LEFT-LEFT, LEFT-INNER, INNER-LEFT) and they
# return 1, 4, 1 and 2 rows. Any two of them agreeing by accident is possible; all four is not.
#
# `SELECT *` across a chain returns every table's columns left to right, in the order **written**,
# which is not necessarily the order the executor reads them.
#
# This file builds and drops its own tables, so replaying it twice is the same as replaying it
# once — verified by capturing it twice and diffing.
#
DROP TABLE IF EXISTS jc		-
DROP TABLE IF EXISTS jb		-
DROP TABLE IF EXISTS ja		-
CREATE TABLE ja (id int8 PRIMARY KEY, name text, bid int8)		-
CREATE TABLE jb (id int8 PRIMARY KEY, label text, cid int8)		-
CREATE TABLE jc (id int8 PRIMARY KEY, tag text)		-
