# **A bare literal compared against a `regclass` is an oid** — PostgreSQL 19beta1, 2026-09-09.
# `ON_ERROR_ROLLBACK on`, one `BEGIN … ROLLBACK`.
#
# `debts-v1.1.md` #41, the comparison half. `=` over a `regclass` is `oideq`, so an unadorned
# literal is handed to `oidin` and a *name* is `22P02` — not resolved. The same rule ADR 0098
# measured for `regproc`, one type over, and it needs no catalog at all.
#
# ## THINGS REASONING WOULD GET WRONG
#
#   * **An assignment is the other way.** `INSERT INTO rh VALUES (1, 'ra')` stores the relation:
#     an assignment goes through `regclassin`, which resolves the name. Measured beside this.
#   * **`IN` is not `=`.** `r IN ('ra','rb')` **answers**, where `r = 'ra'` is `22P02` — the list
#     is coerced through the *type's* input function rather than through the operator's operand
#     type. It is the one shape that would have been assumed from the others.
#   * **`<>` behaves as `=` does**, and a literal of *digits* is an oid and simply compares:
#     `r = '0'` is no rows rather than an error.
#   * `r = 'ra'::regclass` and `r::text = 'ra'` are the two spellings that answer.
#
# ## The measurements
#
CREATE TABLE ra (id int8)		-
CREATE TABLE rb (id int8)		-
CREATE TABLE rh (id int8, r regclass)		-
INSERT INTO rh VALUES (1, 'ra'::regclass), (2, 'rb'::regclass)		-
SELECT id FROM rh WHERE r = 'ra'	!22P02 invalid input syntax for type oid: "ra"
SELECT id FROM rh WHERE 'ra' = r	!22P02 invalid input syntax for type oid: "ra"
SELECT id FROM rh WHERE r = ANY('{ra}')	!22P02 invalid input syntax for type oid: "ra"
SELECT id FROM rh WHERE r IN ('ra','rb') ORDER BY id	bigint	1 ; 2
SELECT id FROM rh WHERE r = 'ra'::regclass	bigint	1
SELECT id FROM rh WHERE r::text = 'ra'	bigint	1
SELECT id FROM rh WHERE r <> 'ra'	!22P02 invalid input syntax for type oid: "ra"
SELECT id FROM rh WHERE r = '0'	bigint	-
DROP TABLE rh		-
DROP TABLE ra		-
DROP TABLE rb		-
