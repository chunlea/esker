# The `pg_type` rows an array column needs before `ActiveRecord` can quote one.
#
# **Run 52 ranking: `TypeError` 43 tests over 2 files, 42 of them `array_test.rb`.** It is a *Ruby*
# error, `can't quote Array` at `postgresql/quoting.rb:122`, and it is not a bug in the adapter: it
# is what the adapter does when it was never told the column is an array. AR registers array types
# from its boot type-map query — the one `boot36` counts — and it looks the array type up **by the
# element type's `typarray` oid**. If that oid has no row, no `OID::Array` is registered, the column
# is an ordinary one as far as the adapter knows, and quoting a Ruby `Array` for it falls through to
# the generic `quote` and raises.
#
# On the node at run 52 that is exactly the state: **6 array types in `pg_type`, and 16 base types
# whose `typarray` points at a row that is not there** — `varchar`, `bool`, `bpchar`, `bytea`,
# `citext`, `date`, `float4`, `float8`, `interval`, `json`, `jsonb`, `oid`, `time`, `timestamp`,
# `timestamptz`, `uuid`. `_int4` and `_text` exist, which is why `t.integer "ratings", array: true`
# works in that file and `t.string "tags", array: true` raises. `array_test.rb` declares
# `string`, `integer`, `datetime`, `hstore`, `decimal` and `timestamp` arrays (`:21-26`).
#
# Things reasoning would get wrong:
#
# * **`typarray` and `typelem` are a two-way link and both halves are load-bearing.** The base row
#   points *up* to its array type; the array row points *down* through `typelem`. A client walks it
#   in both directions — AR from the base type, `format_type` from the array — so filling in one and
#   not the other is worse than filling in neither: it advertises a type that cannot be fetched.
# * **The check is one statement**, and it is the last one below: on this oracle it answers **0**.
#   Any number above zero is a dangling reference.
# * **An array type is not identified by its name.** `_int4` is a convention, not a rule: the test is
#   `typelem <> 0 AND typinput = 'array_in'`. A name-based rule breaks on a user type whose array is
#   named by truncation, and on `citext`, whose array oid is allocated at `CREATE EXTENSION` time and
#   is not a fixed number at all.
# * **`typinput = 'array_in'` does NOT imply `typcategory = 'A'`**, and the row below that asks the
#   question answers `f`. There is exactly one exception on this oracle and it is the pseudo-type
#   `_record`, `typcategory = 'P'`. So a check written as "every array_in type is category A" fails
#   on a real PostgreSQL, and the two columns have to be populated from what each one means.
# * **The delimiter is not always `,`** — `box` uses `;`. Both the element row and the array row
#   carry `typdelim` and on this oracle they **agree**, so these rows do not discriminate which of
#   the two a parser must read; they only settle that a hard-coded comma is wrong.
#
# The whole capture runs inside `BEGIN … ROLLBACK` and touches nothing: every statement reads
# `pg_catalog`. The `CREATE EXTENSION`s at the end are the citext/hstore half of the question and
# roll back with everything else; the `SELECT count(*)` lines prove it.
#
# Rows start with a literal `'r'` because `sesscap.py` drops output rows that begin with a space.
BEGIN		-
# ---- AR's boot type-map query, the array half, over the types the suite declares ----
SELECT 'r', t.oid, t.typname, t.typelem, t.typdelim, t.typinput, t.typtype, t.typbasetype FROM pg_type as t LEFT JOIN pg_range as r ON t.oid = r.rngtypid WHERE t.typname IN ('_int4','_text','_varchar','_timestamp','_timestamptz','_date','_numeric','_uuid','_bool','_jsonb') ORDER BY t.typname	text,oid,name,oid,"char",regproc,"char",oid	r|1000|_bool|16|,|array_in|b|0 ; r|1182|_date|1082|,|array_in|b|0 ; r|1007|_int4|23|,|array_in|b|0 ; r|3807|_jsonb|3802|,|array_in|b|0 ; r|1231|_numeric|1700|,|array_in|b|0 ; r|1009|_text|25|,|array_in|b|0 ; r|1115|_timestamp|1114|,|array_in|b|0 ; r|1185|_timestamptz|1184|,|array_in|b|0 ; r|2951|_uuid|2950|,|array_in|b|0 ; r|1015|_varchar|1043|,|array_in|b|0
# ---- the two-way link, for the types array_test.rb declares ----
SELECT 'r', b.typname AS base, b.typarray, a.typname AS array_name, a.typelem, a.typelem = b.oid AS points_back FROM pg_type b JOIN pg_type a ON a.oid = b.typarray WHERE b.typname IN ('varchar','int4','timestamp','timestamptz','numeric','text','uuid','bool','date') ORDER BY b.typname	text,name,oid,name,oid,boolean	r|bool|1000|_bool|16|t ; r|date|1182|_date|1082|t ; r|int4|1007|_int4|23|t ; r|numeric|1231|_numeric|1700|t ; r|text|1009|_text|25|t ; r|timestamp|1115|_timestamp|1114|t ; r|timestamptz|1185|_timestamptz|1184|t ; r|uuid|2951|_uuid|2950|t ; r|varchar|1015|_varchar|1043|t
# ---- how an array type is recognised without looking at its name ----
SELECT 'r', typname, typcategory, typelem <> 0 AS has_element, typinput FROM pg_type WHERE typname IN ('_int4','_varchar','int4','varchar') ORDER BY typname	text,name,"char",boolean,regproc	r|_int4|A|t|array_in ; r|_varchar|A|t|array_in ; r|int4|N|f|int4in ; r|varchar|S|f|varcharin
SELECT 'r', count(*) FROM pg_type WHERE typinput = 'array_in'::regproc	text,bigint	r|369
SELECT 'r', count(*) FROM pg_type WHERE typcategory = 'A'	text,bigint	r|370
SELECT 'r', count(*) = count(*) FILTER (WHERE typcategory = 'A') AS array_in_implies_category_A FROM pg_type WHERE typinput = 'array_in'::regproc	text,boolean	r|f
# ---- the delimiter comes from the ELEMENT row, and is not always a comma ----
SELECT 'r', typname, typdelim FROM pg_type WHERE typname IN ('int4','text','varchar','box','_box') ORDER BY typname	text,name,"char"	r|_box|; ; r|box|; ; r|int4|, ; r|text|, ; r|varchar|,
SELECT 'r', b.typname, b.typdelim AS element_delim, a.typname AS array_name, a.typdelim AS array_row_delim FROM pg_type b JOIN pg_type a ON a.oid = b.typarray WHERE b.typname = 'box'	text,name,"char",name,"char"	r|box|;|_box|;

# ---- a value of every array type the suite's arrays need, so the parser has a row for each ----
SELECT 'r', ARRAY['a','b']::varchar[], ARRAY[1,2]::int4[], ARRAY['2010-01-01 14:30'::timestamp], ARRAY['2010-01-01'::date]	text,character varying[],integer[],timestamp without time zone[],date[]	r|{a,b}|{1,2}|{"2010-01-01 14:30:00"}|{2010-01-01}
SELECT 'r', ARRAY[1.5,2.5]::numeric[], ARRAY[true,false], ARRAY['{"a":1}'::jsonb], ARRAY['11111111-1111-1111-1111-111111111111'::uuid]	text,numeric[],boolean[],jsonb[],uuid[]	r|{1.5,2.5}|{t,f}|{"{\"a\": 1}"}|{11111111-1111-1111-1111-111111111111}
SELECT 'r', pg_typeof(ARRAY['a','b']::varchar[]), pg_typeof(ARRAY[1.5]::numeric[]), pg_typeof(ARRAY[true])	text,regtype,regtype,regtype	r|character varying[]|numeric[]|boolean[]
# ---- the extension types allocate their array oid when they are installed ----
CREATE EXTENSION IF NOT EXISTS hstore		-
CREATE EXTENSION IF NOT EXISTS citext		-
SELECT 'r', b.typname, b.typarray <> 0 AS has_array, a.typname AS array_name, a.typelem = b.oid AS points_back FROM pg_type b JOIN pg_type a ON a.oid = b.typarray WHERE b.typname IN ('hstore','citext') ORDER BY b.typname	text,name,boolean,name,boolean	r|citext|t|_citext|t ; r|hstore|t|_hstore|t
SELECT 'r', b.typname, b.oid < 10000 AS oid_is_builtin, a.oid < 10000 AS array_oid_is_builtin FROM pg_type b JOIN pg_type a ON a.oid = b.typarray WHERE b.typname IN ('hstore','citext','varchar') ORDER BY b.typname	text,name,boolean,boolean	r|citext|f|f ; r|hstore|f|f ; r|varchar|t|t
SELECT 'r', ARRAY['a=>1'::hstore], ARRAY['ABC'::citext]		r|{"\"a\"=>\"1\""}|{ABC}
# ---- THE CHECK. Every `typarray` must name a row that exists. This answers 0. ----
SELECT 'r', count(*) AS dangling_typarray FROM pg_type t WHERE t.typarray <> 0 AND NOT EXISTS (SELECT 1 FROM pg_type a WHERE a.oid = t.typarray)	text,bigint	r|0
# ---- and its mirror: every array row's typelem must name a row that exists ----
SELECT 'r', count(*) AS dangling_typelem FROM pg_type t WHERE t.typinput = 'array_in'::regproc AND NOT EXISTS (SELECT 1 FROM pg_type e WHERE e.oid = t.typelem)	text,bigint	r|0
# ---- ...and the link must round-trip in both directions ----
SELECT 'r', count(*) AS asymmetric FROM pg_type b JOIN pg_type a ON a.oid = b.typarray WHERE a.typelem <> b.oid	text,bigint	r|0
