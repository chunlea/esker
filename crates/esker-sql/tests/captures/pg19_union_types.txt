# **What type a set operation's column has** — PostgreSQL 19, 2026-09-07, one `BEGIN … ROLLBACK`.
#
# Read through `\gdesc`, not `pg_typeof(...) LIMIT 1`: a `UNION ALL` in a `FROM` is flattened and
# `pg_typeof` then reports each branch's own type, which is how the first probe of this said
# `integer` for `int8 UNION ALL real`.
#
# The rule (`select_common_type`): two categories cannot be matched (42804); a preferred type stays;
# otherwise the later type is taken only when the earlier casts to it implicitly and not the reverse —
# so `varchar` beside `text` stays `character varying`, and `text` beside `varchar` is `text`. Then
# every arm must reach the chosen type implicitly, or it is 42846 naming that arm.
#
# ---- the probes, as sen
# \set ON_ERROR_ROLLBACK on
# \set VERBOSITY terse
# BEGIN;
# \echo === varchar/tex
# SELECT 'a'::varchar(5) AS n UNION ALL SELECT 'b'::text \gdesc
# \echo === text/varchar
# SELECT 'a'::text AS n UNION ALL SELECT 'b'::varchar(5) \gdesc
# \echo === bpchar/tex
# SELECT 'a'::char(2) AS n UNION ALL SELECT 'b'::text \gdesc
# \echo === bpchar/varchar
# SELECT 'a'::char(2) AS n UNION ALL SELECT 'b'::varchar \gdesc
# \echo === varchar/bpchar
# SELECT 'a'::varchar AS n UNION ALL SELECT 'b'::char(2) \gdesc
# \echo === varchar(3)/varchar(5)
# SELECT 'abc'::varchar(3) AS n UNION ALL SELECT 'b'::varchar(5) \gdesc
# \echo === text/unknown

# \echo === unknown/unknown
