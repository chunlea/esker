# **The catalog's two vector shapes** — PostgreSQL 19beta1, 2026-09-09. `ON_ERROR_ROLLBACK on`,
# one `BEGIN … ROLLBACK`, temp table `vt (a int4, b text)` with `CREATE INDEX vt_ab ON vt (a, b)`.
#
# Taken for `pg_index.indkey`, which `ActiveRecord` reads to name an index's columns, and extended
# to `oidvector` because it is the same shape one type over (`pg_proc.proargtypes`, and
# `ARRAY[…]::oidvector` in `case_insensitive_probe.rs`). The corpus beside this file
# (`corpus/pg19_catalog_vectors.txt`) is what replays; these are the rows that were read.
#
#     pg_typeof(indkey)     int2vector    pg_typeof(indkey[0])   smallint
#     pg_typeof(indoption)  int2vector
#     indkey                1 2           indkey[0] 1   indkey[1] 2   array_length(indkey,1) 2
#     indkey::text          1 2           and its type is `text`
#     array_position(indkey, 1::int2)  0            <- **0-based**, unlike an int2[]
#     array_position(indkey, 2::int2)  1
#     attnum = ANY(indkey)  finds a and b
#
#     '1 2'::int2vector     1 2           ('1 2'::int2vector)[0]   1
#     '23 25'::oidvector    23 25         ('23 25'::oidvector)[0]  23
#     '23  25'::oidvector   23 25         <- the spaces are normalised
#     ''::oidvector         (empty)       ''::int2vector           (empty)
#     'abc'::oidvector      !22P02 invalid input syntax for type oid: "abc"
#     'abc'::int2vector     !22P02 invalid input syntax for type smallint: "abc"
#
# **A vector's input function is its element's, applied to every word**, which is why the two
# refusals name `oid` and `smallint` rather than the vector. The catalog rows say the same thing:
#
#     oid  typname     typlen  typtype  typcategory  typdelim  typinput      typelem  typarray
#     22   int2vector  -1      b        A            ,         int2vectorin  21       1006
#     30   oidvector   -1      b        A            ,         oidvectorin   26       1013
#
# `typcategory` is **A** for both and `typelem` names the element — a real server files them with
# the arrays. `typarray` is the one column this node answers differently, and deliberately: see
# the entry in `tests/catalog_vectors.rs`.
#
#     proargtypes of int4in   2275 (cstring)   pg_typeof  oidvector   proargtypes[0]  2275

SELECT 'r', pg_typeof(indkey), pg_typeof(indkey[0]), pg_typeof(indoption) FROM pg_index WHERE indexrelid = 'vt_ab'::regclass	text,regtype,regtype,regtype	r|int2vector|smallint|int2vector
SELECT 'r', array_position(indkey, 1::int2), array_position(indkey, 2::int2) FROM pg_index WHERE indexrelid = 'vt_ab'::regclass	text,integer,integer	r|0|1
SELECT 'r', '23 25'::oidvector, ('23 25'::oidvector)[0], pg_typeof('23 25'::oidvector)	text,oidvector,oid,regtype	r|23 25|23|oidvector
SELECT 'r', '23  25'::oidvector	text,oidvector	r|23 25
SELECT 'r', ''::oidvector, ''::int2vector	text,oidvector,int2vector	r||
SELECT 'r', 'abc'::oidvector	!22P02 invalid input syntax for type oid: "abc"
SELECT 'r', 'abc'::int2vector	!22P02 invalid input syntax for type smallint: "abc"
