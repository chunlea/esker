# `regtype` — as PostgreSQL 19beta1 answers it, which is **an oid that prints as a name**.
#
# Captured 2026-09-05 against 127.0.0.1:55432 in one `BEGIN … ROLLBACK`, unaligned with
# `fieldsep '|'`. This is the spec for ADR 0077 and for the three-gap census behind run 92's six
# ARRAY rows — `ActiveRecord`'s `can_perform_case_insensitive_comparison_for?`
# (`postgresql_adapter.rb:1081`), whose answer is `r12`/`r13`/`r14`.
#
# Five things reasoning would get wrong, and `r5` is the one that decides the model:
#
#   * **A `regtype` orders by its oid, not by its name.** `'text'::regtype < 'int4'::regtype` is
#     **false**, because 25 < 23 is false — alphabetically `int4` would sort first. So the value is
#     the number and the name is only how it prints.
#   * **`'text'::regtype = 25` is true.** It compares against a bare integer with no cast, which no
#     text-shaped model can reproduce.
#   * **An oid that is not a type prints as the number and is not an error** (`r16`), while a
#     *name* that is not a type is `42704` (`r15`). The two directions do not match.
#   * **`::oidvector` prints digits, not names** (`r7`): `ARRAY['text'::regtype]::oidvector` is
#     `25`, which is exactly what `proargtypes` holds (`r8`).
#   * `'int4'::regtype` prints `integer` — the **canonical** name, not the spelling written.
#
# `r14` is the control: `lower(integer)` does not exist and `integer` does not cast to anything
# `lower` takes, so the probe is **false** for an integer column. A model that answered `t` to
# everything would pass `r12` and `r13` and be wrong.

BEGIN;		-
SELECT 'r1', 'integer'::regtype, 'int4'::regtype, 'character varying'::regtype, 'text'::regtype;		r1|integer|integer|character varying|text
SELECT 'r2', 'integer'::regtype::oid, 'text'::regtype::oid, 'character varying'::regtype::oid;		r2|23|25|1043
SELECT 'r3', pg_typeof('integer'::regtype), pg_typeof('integer'::regtype::oid);		r3|regtype|oid
SELECT 'r4', 23::regtype, 25::regtype, 1043::regtype;		r4|integer|text|character varying
SELECT 'r5', 'text'::regtype = 25, 'text'::regtype = 'text'::regtype, 'text'::regtype < 'int4'::regtype;		r5|t|t|f
SELECT 'r6', ARRAY['text'::regtype], pg_typeof(ARRAY['text'::regtype]);		r6|{text}|regtype[]
SELECT 'r7', ARRAY['text'::regtype]::oidvector, pg_typeof(ARRAY['text'::regtype]::oidvector);		r7|25|oidvector
SELECT 'r8', proargtypes, pg_typeof(proargtypes) FROM pg_proc WHERE proname='lower' ORDER BY oid;		r8|25|oidvector ; r8|3831|oidvector ; r8|4537|oidvector
SELECT 'r9', proargtypes = ARRAY['text'::regtype]::oidvector FROM pg_proc WHERE proname='lower' ORDER BY oid;		r9|t ; r9|f ; r9|f
SELECT 'r10', castsource, casttarget, pg_typeof(castsource) FROM pg_cast WHERE castsource='character varying'::regtype ORDER BY casttarget LIMIT 4;		r10|1043|18|oid ; r10|1043|19|oid ; r10|1043|25|oid ; r10|1043|142|oid
SELECT 'r11', count(*) FROM pg_proc p INNER JOIN pg_cast c ON ARRAY[c.casttarget]::oidvector = p.proargtypes WHERE p.proname='lower' AND c.castsource='character varying'::regtype;		r11|1
-- ------------------------------------------------- the probe itself, for three column types		-
SELECT 'r12', exists(SELECT * FROM pg_proc WHERE proname='lower' AND proargtypes = ARRAY['character varying'::regtype]::oidvector) OR exists(SELECT * FROM pg_proc INNER JOIN pg_cast ON ARRAY[casttarget]::oidvector = proargtypes WHERE proname='lower' AND castsource='character varying'::regtype);		r12|t
SELECT 'r13', exists(SELECT * FROM pg_proc WHERE proname='lower' AND proargtypes = ARRAY['text'::regtype]::oidvector) OR exists(SELECT * FROM pg_proc INNER JOIN pg_cast ON ARRAY[casttarget]::oidvector = proargtypes WHERE proname='lower' AND castsource='text'::regtype);		r13|t
SELECT 'r14', exists(SELECT * FROM pg_proc WHERE proname='lower' AND proargtypes = ARRAY['integer'::regtype]::oidvector) OR exists(SELECT * FROM pg_proc INNER JOIN pg_cast ON ARRAY[casttarget]::oidvector = proargtypes WHERE proname='lower' AND castsource='integer'::regtype);		r14|f
-- ------------------------------------------------- a name that is not a type, and an oid that is not		-
SAVEPOINT s1;		-
SELECT 'r15', 'nosuchtype'::regtype;	!42704 type "nosuchtype" does not exist
ROLLBACK TO s1;		-
SELECT 'r16', 999999::regtype;		r16|999999
SELECT 'r17', 999999::regtype::text;		r17|999999
ROLLBACK;		-
