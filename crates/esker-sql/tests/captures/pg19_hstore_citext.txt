# `hstore` and `citext` as COLUMN TYPES and VALUES — what `hstore_test.rb` and `citext_test.rb` send.
#
# **Run 47 ranking row 7: `extension "…" is not available`, 114 tests over 9 files**, and hstore is
# the bulk of it. `captures/pg19_extensions.txt` already pins the *install* mechanics — availability,
# `CREATE`/`DROP EXTENSION`, schemas, the two error probes. This file is the other half and does not
# repeat it: **the two types once they are installed**, which is every statement in the two test
# files that reaches the server.
#
# Things reasoning would get wrong:
#
# * **`hstore` and `citext` are not built-in oids.** They are allocated when the extension is
#   installed, so nothing may hard-code them; AR's boot type-map query reads them by `typname` and
#   the column reader gets the name back through `format_type`. Both queries are below verbatim.
# * **`citext` is not `text` with a `lower()` around it.** It is its own type with its own equality,
#   and `test_case_insensitiveness` asserts AR emits **no** `lower(...)` for it — so `=` itself must
#   be case-insensitive, and `ORDER BY`, `DISTINCT`, `LIKE` and `~~` follow that collation too. The
#   rows below say which of them fold case and which do not, because they are not all the same.
# * **`hstore` has an array form.** `hstore_test.rb` declares `t.hstore "payload", array: true`, so
#   `hstore[]` has to parse and render, including a NULL *value inside* a hstore inside an array —
#   which is not the same thing as a NULL element.
# * **The quoting is the test suite's, not a tidy example.** `test_backslash`, `test_quotes`,
#   `test_parse6/7` and `test_comma` push `"`, `\`, `,` and `=>` through the value round trip; those
#   exact pairs are below. A parser that handles `a=>b` and not these passes no test in that file.
# * **`t.hstore "tags", default: ""`** — the empty hstore is a *default*, and an empty hstore is not
#   NULL and not the empty string when it comes back.
#
# The whole capture runs inside `BEGIN … ROLLBACK`; error probes sit in `SAVEPOINT`/`ROLLBACK TO`,
# and the `SELECT count(*)` lines after the `ROLLBACK` prove it left nothing on the oracle. Because
# `\gdesc` runs in a second session *after* that rollback, every statement naming `hst` or `cit` has
# an **empty** declared-types column; the `pg_typeof` lines are what pin the types instead.
#
# Rows start with a literal `'r'` because `sesscap.py` drops output rows that begin with a space.
BEGIN		-
# ---- what the adapter asks before it uses either type (postgresql_adapter.rb:488-530) ----
SELECT 'r', true FROM pg_available_extensions WHERE name = 'hstore'	text,boolean	r|t
SELECT 'r', installed_version IS NOT NULL FROM pg_available_extensions WHERE name = 'hstore'	text,boolean	r|f
CREATE EXTENSION IF NOT EXISTS "hstore"		-
CREATE EXTENSION IF NOT EXISTS "citext"		-
SELECT 'r', extname, extversion FROM pg_extension WHERE extname IN ('citext','hstore') ORDER BY extname	text,name,text	r|citext|1.8 ; r|hstore|1.8
# ---- AR's boot type-map query, narrowed to the two names (the oids are assigned, never fixed) ----
SELECT 'r', t.typname, t.typelem, t.typdelim, t.typinput, t.typtype, t.typbasetype, t.typcategory, t.typlen FROM pg_type as t WHERE t.typname IN ('hstore','citext') ORDER BY t.typname	text,name,oid,"char",regproc,"char",oid,"char",smallint	r|citext|0|,|citextin|b|0|S|-1 ; r|hstore|0|,|hstore_in|b|0|U|-1
SELECT 'r', t.typname, t.oid = 0 AS oid_is_zero, t.typarray = 0 AS no_array_type FROM pg_type t WHERE t.typname IN ('hstore','citext') ORDER BY t.typname	text,name,boolean,boolean	r|citext|f|f ; r|hstore|f|f
SELECT 'r', pg_typeof('a=>b'::hstore), pg_typeof('x'::citext), pg_typeof(ARRAY['a=>b'::hstore])		r|hstore|citext|hstore[]
# ---- the tables the two files create in setup ----
DROP TABLE IF EXISTS hst		-

# ---- column_definitions(table) verbatim (postgresql_adapter.rb:1057) — how AR learns the type ----
SELECT 'r', a.attname, format_type(a.atttypid, a.atttypmod), pg_get_expr(d.adbin, d.adrelid), a.attnotnull, a.atttypmod, c.collname, col_description(a.attrelid, a.attnum) AS comment, attidentity AS identity, attgenerated as attgenerated FROM pg_attribute a LEFT JOIN pg_attrdef d ON a.attrelid = d.adrelid AND a.attnum = d.adnum LEFT JOIN pg_type t ON a.atttypid = t.oid LEFT JOIN pg_collation c ON a.attcollation = c.oid AND a.attcollation <> t.typcollation WHERE a.attrelid = '"hst"'::regclass AND a.attnum > 0 AND NOT a.attisdropped ORDER BY a.attnum		r|id|integer|nextval('hst_id_seq'::regclass)|t|-1|\N|\N|| ; r|tags|hstore|''::hstore|f|-1|\N|\N|| ; r|payload|hstore[]|\N|f|-1|\N|\N|| ; r|settings|hstore|\N|f|-1|\N|\N||
SELECT 'r', a.attname, format_type(a.atttypid, a.atttypmod), pg_get_expr(d.adbin, d.adrelid), a.attnotnull FROM pg_attribute a LEFT JOIN pg_attrdef d ON a.attrelid = d.adrelid AND a.attnum = d.adnum WHERE a.attrelid = '"cit"'::regclass AND a.attnum > 0 AND NOT a.attisdropped ORDER BY a.attnum		r|id|integer|nextval('cit_id_seq'::regclass)|t ; r|cival|citext|\N|f

# ---- the operators and functions the file and the adapter reach for ----
SELECT 'r', ('a=>b, c=>d'::hstore)->'a', ('a=>b'::hstore) ? 'a', ('a=>b, c=>d'::hstore) @> 'a=>b'		r|b|t|t

# ---- hstore[] — `t.hstore "payload", array: true` ----

# A unique index over citext FOLDS CASE, and that is the whole reason this section is here:
# `validates_uniqueness_of` on a citext column is enforced by one, and two rows that differ only in
# case cannot both be in it. The index cannot even be built over the two rows above -- that refusal
# is the evidence, so it is an error probe rather than an accident.
SAVEPOINT s3		-
