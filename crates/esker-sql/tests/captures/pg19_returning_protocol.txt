# The extended protocol's result contract, for the statement shapes behind run 54's `UnableToSend`.
#
# **Run 54 ranking: `ActiveRecord::StatementInvalid` with no server text, 19 tests over 7 files.**
# The dig found 17 of the 19 are one message and it is not a SQL answer at all:
#
#     PG::UnableToSend: server sent data ("D" message) without prior row description ("T" message)
#
# — libpq refusing this node's message stream, raised from `exec_params`
# (`postgresql/database_statements.rb:169`). One more is `unexpected field count in "D" message`,
# the same fault one field along. The files: `infinity_test.rb` (6), `bytea_test.rb` (4),
# `numbers_test.rb` (3), `uuid_test.rb` (2), `check_constraint_test.rb` (2),
# `postgresql_adapter_test.rb`, `persistence_test.rb`.
#
# The statement is always the same shape — `INSERT INTO "…" ("col") VALUES ($1) RETURNING "id"` with
# a bind — and in `bytea_test.rb` the sequence around it repeats **exactly four times**:
#
#     1. a server error inside a transaction   (`relation "…" does not exist`)
#     2. the next statement, correctly refused  (`25P02 current transaction is aborted`)
#     3. the INSERT … RETURNING                 -> D without T
#
# **It is the same class as run 54's other wire finding**, `message contents do not agree with
# length in message type "E"`: both are the node's framing going out of step on a connection, one
# on the error message, one on the row description. Neither is reduced to a minimal repro. Five
# shapes were tried against the node and none of them desynchronised it, and they are listed in
# `results/run-55.md` so nobody repeats them.
#
# An oracle cannot show a bug it does not have. What it can pin is **what must come back**, and the
# `types` column below is exactly that: it is the RowDescription. A row with types is a statement
# that MUST send a `T` before any `D`; a row with an empty types column and `-` is a command tag
# that must send **no** `D` at all. The two rows to read hardest are the `RETURNING` statements that
# match **nothing**: one field, zero rows — a `T` and then no `D`, which is the case an
# implementation is most likely to get wrong in the direction run 54 saw.
DROP TABLE IF EXISTS rp	[]		-
CREATE TABLE rp (id serial primary key, payload bytea, amount numeric(10,2), ratio double precision, tag uuid, n integer)	[]		-
# ---- a command tag: no T, no D ----
INSERT INTO "rp" ("n") VALUES ($1)	[1]		-
UPDATE "rp" SET "n" = $1 WHERE "id" = $2	[2, 1]		-
DELETE FROM "rp" WHERE "id" = $1	[999]		-
# ---- RETURNING: a T, then one D ----
INSERT INTO "rp" ("n") VALUES ($1) RETURNING "id"	[3]	integer	2
INSERT INTO "rp" DEFAULT VALUES RETURNING "id"	[]	integer	3
INSERT INTO "rp" ("payload") VALUES ($1) RETURNING "id"	["\\x68656c6c6f20776f726c64"]	integer	4
INSERT INTO "rp" ("amount") VALUES ($1) RETURNING "id", "amount"	["12.34"]	integer,numeric	5|12.34
INSERT INTO "rp" ("ratio") VALUES ($1) RETURNING "id", "ratio"	["Infinity"]	integer,double precision	6|Infinity
INSERT INTO "rp" ("ratio") VALUES ($1) RETURNING "id", "ratio"	["-Infinity"]	integer,double precision	7|-Infinity
INSERT INTO "rp" ("ratio") VALUES ($1) RETURNING "id", "ratio"	["NaN"]	integer,double precision	8|NaN
INSERT INTO "rp" ("tag") VALUES ($1) RETURNING "id", "tag"	["11111111-1111-1111-1111-111111111111"]	integer,uuid	9|11111111-1111-1111-1111-111111111111
# ---- THE ONES TO READ HARDEST: RETURNING that matches nothing. One field, zero rows. ----
UPDATE "rp" SET "n" = $1 WHERE "id" = $2 RETURNING "id"	[7, 999999]	integer	-
DELETE FROM "rp" WHERE "id" = $1 RETURNING "id"	[999999]	integer	-
SELECT "rp"."id" FROM "rp" WHERE "rp"."id" = $1	[999999]	integer	-
# ---- a SELECT that matches, for the contrast ----
SELECT "rp"."id", "rp"."n" FROM "rp" WHERE "rp"."n" = $1	[3]	integer,integer	2|3
SELECT count(*) FROM "rp"	[]	bigint	9
# ---- one more shape from the failing files: a RETURNING whose column list is not the pk ----
INSERT INTO "rp" ("n", "amount") VALUES ($1, $2) RETURNING "amount", "n"	[11, "9.99"]	numeric,integer	9.99|11
UPDATE "rp" SET "n" = $1 RETURNING "id", "n"	[12]	integer,integer	1|12 ; 2|12 ; 3|12 ; 4|12 ; 5|12 ; 6|12 ; 7|12 ; 8|12 ; 9|12 ; 10|12
# ---- an error through the extended protocol still carries its SQLSTATE and no fields ----
SELECT * FROM nosuchtable_rp WHERE 1 = $1	[1]		!42P01 relation "nosuchtable_rp" does not exist
INSERT INTO "rp" ("tag") VALUES ($1) RETURNING "id"	["not-a-uuid"]		!22P02 invalid input syntax for type uuid: "not-a-uuid"
SELECT count(*) FROM "rp"	[]	bigint	10
#
# **The error-ordering sequence is deliberately NOT in this file, and cannot be.** `bindcap.rb`
# wraps the whole run in one transaction and every probe in its own `SAVEPOINT`/`ROLLBACK TO`, the
# same isolation `two-server-replay.rb --isolate` uses and for the same reason: without it the first
# refusal turns every later probe into `25P02` and the file stops being a measurement. The cost is
# that a probe cannot observe an *aborted* transaction, because the savepoint has already undone the
# abort — the first cut of this file tried, and its `BEGIN … error … ROLLBACK` block silently
# recorded PostgreSQL *not* aborting, which is false. What the node does after an error is a
# session-sequence question, and the evidence for it is
# `triage/stopped-6b6c276/transactions_test.rb.out` and this round's ruled-out repros, not a corpus
# row.
